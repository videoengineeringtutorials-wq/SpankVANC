cmake_minimum_required(VERSION 3.16)
project(SpankVANCNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(check-anc check_anc.cpp)
add_test(NAME anc-policy COMMAND check-anc)
foreach(BOARD ESP32 ESP8266)
  add_executable(check-firmware-${BOARD} ../tests/check_firmware.cpp)
  target_include_directories(check-firmware-${BOARD} PRIVATE ../tests/arduino_stub)
  target_compile_definitions(check-firmware-${BOARD} PRIVATE ${BOARD})
  add_test(NAME firmware-${BOARD} COMMAND check-firmware-${BOARD})
endforeach()
set(DECKLINK_SDK_INCLUDE "" CACHE PATH "DeckLink SDK Linux/include directory")
if(DECKLINK_SDK_INCLUDE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The diagnostic currently uses the Linux SDK. Windows COM adapter is planned.")
  endif()
  add_executable(spankvanc-decklink decklink_anc.cpp "${DECKLINK_SDK_INCLUDE}/DeckLinkAPIDispatch.cpp")
  target_include_directories(spankvanc-decklink PRIVATE "${DECKLINK_SDK_INCLUDE}")
  target_compile_options(spankvanc-decklink PRIVATE -Wall -Wextra -Wno-multichar)
  target_link_libraries(spankvanc-decklink PRIVATE dl pthread)
endif()
add_executable(check-scte check_scte.cpp)
